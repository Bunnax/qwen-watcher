<script>
/* QWEN WATCHER Telegram publisher (tail module v1.1, self-logging).
   Reads existing canonical frontend state ONLY; validates before publishing;
   logs send/skip reasons into the diagnostics drawer via dbg(). */
(function () {
  var lastSig = {};
  var lastPlanSig = '';
  var lastSkipLog = {};
  var announced = {};
  function fresh(d, ms) { return d && d.lastUpdate != null && (Date.now() - d.lastUpdate) < ms; }
  function skip(k, why) {
    var now = Date.now();
    if (!lastSkipLog[k] || now - lastSkipLog[k] > 60000) {
      lastSkipLog[k] = now;
      try { dbg('warn', 'PUBLISHER skip ' + k + ': ' + why); } catch (e) {}
    }
  }
  function reason(k) {
    var d = marketData[k];
    if (!d || d.price == null) return 'no price yet';
    if (k === state.asset) {
      if (!state.analysis) return 'no analysis yet';
      if (d.candles.length < 60) return 'candles ' + d.candles.length + ' < 60';
      if (Feed.state !== 'ONLINE' && Feed.state !== 'REST') return 'feed state ' + Feed.state;
      if (!fresh(d, 30000)) return 'stale lastUpdate ' + Math.round((Date.now() - d.lastUpdate) / 1000) + 's';
      return 'unknown';
    }
    if (!d.watchSig || !d.watchSig.signal) return 'no watchSig yet';
    if (d.candles.length < 60) return 'candles ' + d.candles.length + ' < 60';
    if (!fresh(d, 90000)) return 'stale lastUpdate ' + Math.round((Date.now() - d.lastUpdate) / 1000) + 's';
    return 'unknown';
  }
  function snap(k) {
    var d = marketData[k];
    if (!d || d.price == null) return null;
    var now = Date.now();
    if (k === state.asset) {
      var a = state.analysis;
      if (!a) return null;
      if (d.candles.length < 60) return null;
      if (Feed.state !== 'ONLINE' && Feed.state !== 'REST') return null;
      if (!fresh(d, 30000)) return null;
      return { ts: now, asset: k, tf: state.tfId, price: d.price, pct: d.change24h == null ? null : d.change24h,
        signal: a.signal, confidence: a.confidence, entryLow: a.entryLow, entryHigh: a.entryHigh, stop: a.stop, target: a.target,
        regime: a.regime, strength: a.strength, volLabel: a.volLabel, momentum: a.momentum,
        planStatus: (state.plan && state.plan.windows && state.plan.windows[0]) ? state.plan.windows[0].status : null,
        dataQuality: 'ok' };
    }
    var w = d.watchSig;
    if (!w || !w.signal) return null;
    if (d.candles.length < 60) return null;
    if (!fresh(d, 90000)) return null;
    return { ts: now, asset: k, tf: state.tfId, price: d.price, pct: d.change24h == null ? null : d.change24h,
      signal: w.signal, confidence: w.conf, entryLow: null, entryHigh: null, stop: null, target: null,
      regime: null, strength: null, volLabel: null, momentum: null, planStatus: null, dataQuality: 'ok-partial' };
  }
  function publish() {
    var ws = Feed.ws;
    if (!ws || ws.readyState !== 1) { skip('ALL', 'no open /feed socket yet'); return; }
    for (var i = 0; i < ASSET_KEYS.length; i++) {
      var k = ASSET_KEYS[i];
      var s = snap(k);
      if (!s) { skip(k, reason(k)); continue; }
      try {
        s.type = 'qwen-state'; s.v = 1; s.publisher = 'dashboard';
        ws.send(JSON.stringify(s));
        if (!announced[k]) { announced[k] = true; try { dbg('ok', 'PUBLISHER live: sending ' + k + ' snapshots every 5s'); } catch (e) {} }
        if (lastSig[k] && lastSig[k] !== s.signal) {
          ws.send(JSON.stringify({ type: 'qwen-event', v: 1, ts: Date.now(), asset: k, event: 'signal-flip', from: lastSig[k], to: s.signal, confidence: s.confidence }));
        }
        lastSig[k] = s.signal;
      } catch (e) { skip(k, 'send error ' + e.message); }
    }
  }
  function watchPlan() {
    var p = state.plan;
    if (!p || !p.windows) return;
    var sig = p.windows.map(function (w) { return w.status; }).join(',');
    if (lastPlanSig && sig !== lastPlanSig) {
      var inv = null;
      for (var i = 0; i < p.windows.length; i++) if (p.windows[i].status === 'INVALIDATED') inv = p.windows[i];
      var ws = Feed.ws;
      if (inv && ws && ws.readyState === 1) {
        try { ws.send(JSON.stringify({ type: 'qwen-event', v: 1, ts: Date.now(), asset: state.asset, event: 'window-invalidated', from: inv.dir, confidence: inv.confidence })); } catch (e) {}
      }
    }
    lastPlanSig = sig;
  }
  setInterval(publish, 5000);
  setInterval(watchPlan, 2000);
})();
</script>
